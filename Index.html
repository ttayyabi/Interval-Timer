import SwiftUI import WebKit struct WebView: UIViewRepresentable { func makeUIView(context: Context) -> WKWebView { let webConfiguration = WKWebViewConfiguration() // Allows inline media playback and Web APIs webConfiguration.allowsInlineMediaPlayback = true let webView = WKWebView(frame: .zero, configuration: webConfiguration) // Load local index.html file if let htmlPath = Bundle.main.path(forResource: "index", ofType: "html") { let htmlUrl = URL(fileURLWithPath: htmlPath) webView.loadFileURL(htmlUrl, allowingReadAccessTo: htmlUrl.deletingLastPathComponent()) } webView.scrollView.isScrollEnabled = false // Keep fixed app UI return webView } func updateUIView(_ uiView: WKWebView, context: Context) {} } struct ContentView: View { var body: some View { WebView() .edgesIgnoringSafeArea(.all) } }
